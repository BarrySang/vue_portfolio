<script setup>
import { computed, ref, onMounted } from 'vue'
import { getRepos } from '../data/portfolio.js'
import SectionToggle from './SectionToggle.vue'

const repos = ref([])
const error = ref(null)
const loading = ref(true)
const selectedLanguage = ref('all')

const languageOptions = computed(() => {
  const languages = [...new Set(repos.value.map(repo => repo.language).filter(Boolean))]
    .sort((a, b) => a.localeCompare(b, undefined, { sensitivity: 'base' }))

  return [
    ...languages.map(language => ({ label: language, value: `language:${language}` })),
    ...(repos.value.some(repo => !repo.language)
      ? [{ label: 'No language specified', value: 'no-language' }]
      : [])
  ]
})

const filteredRepos = computed(() => {
  if (selectedLanguage.value === 'all') {
    return repos.value
  }

  if (selectedLanguage.value === 'no-language') {
    return repos.value.filter(repo => !repo.language)
  }

  const language = selectedLanguage.value.slice('language:'.length)
  return repos.value.filter(repo => repo.language === language)
})

function formatRepoName(name) {
  const sentenceCaseName = name.toLowerCase()
  return sentenceCaseName.charAt(0).toUpperCase() + sentenceCaseName.slice(1)
}

onMounted(() => {
  getRepos()
    .then(data => {
      repos.value = data.sort((a, b) => new Date(b.updated_at) - new Date(a.updated_at))
    })
    .catch(err => {
      error.value = err.message || String(err)
    })
    .finally(() => {
      loading.value = false
    })
})
</script>

<script setup>
import { computed, nextTick, onBeforeUnmount, onMounted, ref, watch } from 'vue'
import { getWorkExperience } from '../data/portfolio.js'

const TRACKS = {
  mechanical: {
    badge: 'Mechanical & Technical Role',
    drawerBadge: 'Mechanical Track',
    badgeClass:
      'border-amber-500/50 bg-amber-500/10 text-amber-700 dark:text-amber-300',
    dotClass: 'bg-amber-500',
    selectedCardClass:
      'border-amber-500 bg-amber-500/10 shadow-[0_0_18px_rgba(245,158,11,0.25)]',
    orgClass: 'text-amber-700 dark:text-amber-300'
  },
  software: {
    badge: 'Software Development & IT Role',
    drawerBadge: 'Software / IT Track',
    badgeClass:
      'border-cyan-500/50 bg-cyan-500/10 text-cyan-700 dark:text-cyan-300',
    dotClass: 'bg-cyan-500',
    selectedCardClass:
      'border-cyan-500 bg-cyan-500/10 shadow-[0_0_18px_rgba(6,182,212,0.25)]',
    orgClass: 'text-cyan-700 dark:text-cyan-300'
  }
}

const filters = [
  { value: 'all', label: 'All Tracks' },
  { value: 'mechanical', label: 'Mechanical / Tech' },
  { value: 'software', label: 'Software / IT' }
]

const entries = getWorkExperience().map((item, id) => ({
  ...item,
  id,
  year: item.duration.match(/\d{4}/)?.[0] ?? '',
  trackInfo: TRACKS[item.track] ?? TRACKS.mechanical
}))

const activeFilter = ref('all')
const selectedId = ref(entries[0]?.id ?? 0)
const drawerOpen = ref(false)

const scroller = ref(null)
const closeButton = ref(null)
const thumbWidth = ref(100)
const thumbLeft = ref(0)

const cardEls = {}
let lockUntil = 0
let rafId = 0
let triggerEl = null

const visibleEntries = computed(() =>
  activeFilter.value === 'all'
    ? entries
    : entries.filter(entry => entry.track === activeFilter.value)
)

const selected = computed(
  () => visibleEntries.value.find(entry => entry.id === selectedId.value) ?? visibleEntries.value[0]
)

function prefersReducedMotion() {
  return window.matchMedia?.('(prefers-reduced-motion: reduce)').matches
}

function setCardRef(id, el) {
  if (el) cardEls[id] = el
}

function updateProgress() {
  const el = scroller.value
  if (!el || !el.scrollWidth) return
  const ratio = el.clientWidth / el.scrollWidth
  thumbWidth.value = Math.min(100, ratio * 100)
  thumbLeft.value = ratio >= 1 ? 0 : (el.scrollLeft / el.scrollWidth) * 100
}

function syncSelection() {
  const el = scroller.value
  const list = visibleEntries.value
  if (!el || !list.length || Date.now() < lockUntil) return

  let target
  if (el.scrollLeft <= 1) {
    target = list[0]
  } else if (el.scrollLeft + el.clientWidth >= el.scrollWidth - 1) {
    target = list[list.length - 1]
  } else {
    const center = el.scrollLeft + el.clientWidth / 2
    let best = Infinity
    for (const entry of list) {
      const card = cardEls[entry.id]
      if (!card) continue
      const distance = Math.abs(card.offsetLeft + card.offsetWidth / 2 - center)
      if (distance < best) {
        best = distance
        target = entry
      }
    }
  }
  if (target) selectedId.value = target.id
}

function onScroll() {
  cancelAnimationFrame(rafId)
  rafId = requestAnimationFrame(() => {
    updateProgress()
    syncSelection()
  })
}

function centerCard(id) {
  const el = scroller.value
  const card = cardEls[id]
  if (!el || !card) return
  el.scrollTo({
    left: card.offsetLeft - (el.clientWidth - card.offsetWidth) / 2,
    behavior: prefersReducedMotion() ? 'auto' : 'smooth'
  })
}

function selectEntry(id) {
  selectedId.value = id
  lockUntil = Date.now() + 800
  centerCard(id)
}

function setFilter(value) {
  activeFilter.value = value
}

watch(activeFilter, async () => {
  const first = visibleEntries.value[0]
  if (first) selectedId.value = first.id
  lockUntil = Date.now() + 300
  await nextTick()
  scroller.value?.scrollTo({ left: 0, behavior: 'auto' })
  updateProgress()
})

async function openDrawer(id, event) {
  selectedId.value = id
  triggerEl = event?.currentTarget ?? null
  drawerOpen.value = true
  await nextTick()
  closeButton.value?.focus()
}

function closeDrawer() {
  drawerOpen.value = false
  triggerEl?.focus?.()
  triggerEl = null
}

function onKeydown(event) {
  if (event.key === 'Escape' && drawerOpen.value) closeDrawer()
}

watch(drawerOpen, open => {
  document.body.style.overflow = open ? 'hidden' : ''
})

onMounted(() => {
  updateProgress()
  window.addEventListener('resize', updateProgress)
  window.addEventListener('keydown', onKeydown)
})

onBeforeUnmount(() => {
  cancelAnimationFrame(rafId)
  window.removeEventListener('resize', updateProgress)
  window.removeEventListener('keydown', onKeydown)
  document.body.style.overflow = ''
})
</script>
